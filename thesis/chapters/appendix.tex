\chapter{Algorithmic Details}
\label{app:algorithmic-details}

This appendix summarizes the two implementation-level procedures that are most relevant for reproducing the method, namely bounded embedding-guided $A^*$ search and the construction of search-aligned training quadruples. Low-level implementation details are omitted for brevity and documented in the accompanying source repository.\footnote{\url{https://git.webis.de/code-teaching/theses/thesis-kiafard.git}}

\begin{algorithm}[H]
\caption{Bounded embedding-guided $A^*$ search}
\label{alg:bounded-astar}
\begin{algorithmic}[1]
\Require Directed graph $G=(V,E)$, source node $a$, target node $b$, embedding function $\phi$, distance function $d$, visit budget $\tau$
\Ensure A path from $a$ to $b$ if found, otherwise $\emptyset$
\State $open \gets$ priority queue containing $(0,0,a)$
\State $best\_g[a] \gets 0$
\State $parent[a] \gets \textsc{None}$
\State $closed \gets \emptyset$
\While{$open$ is not empty}
    \State $(f,g,u) \gets$ entry with smallest priority from $open$
    \If{$u=b$}
        \State \Return reconstructed path from $a$ to $b$
    \EndIf
    \If{$u \in closed$}
        \State \textbf{continue}
    \EndIf
    \State $closed \gets closed \cup \{u\}$
    \If{$\tau \neq -1$ \textbf{and} $|closed| > \tau$}
        \State \Return $\emptyset$
    \EndIf
    \ForAll{$v \in N^{+}(u)$ with $v \notin closed$}
        \State $g' \gets g + d(\phi(u),\phi(v))$
        \If{$v \notin best\_g$ \textbf{or} $g' < best\_g[v]$}
            \State $best\_g[v] \gets g'$
            \State $parent[v] \gets u$
            \State $h' \gets d(\phi(v),\phi(b))$
            \State insert $(g'+h',g',v)$ into $open$
        \EndIf
    \EndFor
\EndWhile
\State \Return $\emptyset$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Construction of search-aligned training quadruples}
\label{alg:training-dataset}
\begin{algorithmic}[1]
\Require Causal training pairs $Q$, causal graph $G=(V,E)$, base embedding model $M$, distance function $d$
\Ensure Training set $D$ of quadruples $(c,e,p,n)$
\State Define the embedding function $\phi(v) \gets M(v)$
\State Precompute embeddings $\mathbf{z}_v \gets \phi(v)$ for all $v \in V$
\State $D \gets \emptyset$
\ForAll{$(A,B) \in Q$}
    \State Let $a,b\in V$ be the graph nodes corresponding to $A,B$
    \State $P \gets$ path returned by $A^*$ on $G$ from $a$ to $b$ using $\phi$, $d$, and $\tau=-1$
    \If{$P=\emptyset$ or $|P| < 2$}
        \State \textbf{continue}
    \EndIf
    \For{$i \gets 0$ to $|P|-2$}
        \State $c \gets P_i$
        \State $p \gets P_{i+1}$
        \State $e \gets b$
        \ForAll{$n \in N^+(c) \setminus \{p\}$}
            \State $D \gets D \cup \{(c,e,p,n)\}$
        \EndFor
    \EndFor
\EndFor
\State \Return $D$
\end{algorithmic}
\end{algorithm}
